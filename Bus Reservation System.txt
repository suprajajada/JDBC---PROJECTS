create database bus_reservation;

use bus_reservation;

create table buses (
    bus_id int primary key,
    bus_name varchar(50),
    source varchar(50),
    destination varchar(50),
    total_seats int,
    available_seats int
);

create table passengers (
    passenger_id int primary key auto_increment,
    passenger_name varchar(50),
    age int,
    gender varchar(10),
    phone varchar(15)
);

create table bookings (
    booking_id int primary key auto_increment,
    bus_id int,
    passenger_id int,
    seat_number int,
    booking_date timestamp default current_timestamp,
    status varchar(20),
    foreign key (bus_id) references buses(bus_id),
    foreign key (passenger_id) references passengers(passenger_id)
);

import java.sql.Connection;

import java.sql.ResultSet;

public class BusDAO {

    public void searchBuses(String source,
                            String destination) {

        String sql =
                "select * from buses " +
                "where source = ? and destination = ?";

        try (Connection con =
                     DBConnection.getConnection();
             PreparedStatement ps =
                     con.prepareStatement(sql)) {

            ps.setString(1, source);
            ps.setString(2, destination);

            ResultSet rs = ps.executeQuery();

            boolean found = false;

            System.out.println(
                    "\n========== BUS SEARCH ==========");

            while (rs.next()) {

                found = true;

                System.out.println(
                        "\nBus ID          : " +
                        rs.getInt("bus_id"));

                "select available_seats " +
                "from buses " +
                "where bus_id = ? for update";

        PreparedStatement ps =
                con.prepareStatement(sql);

        ps.setInt(1, busId);

        ResultSet rs = ps.executeQuery();

        if (rs.next()) {
            return rs.getInt("available_seats");

        ps.executeUpdate();
    }

                "insert into passengers " +
                "(passenger_name, age, gender, phone) " +

                con.prepareStatement(

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.Statement;

public class BookingDAO {

    public boolean isSeatBooked(Connection con,
                                int busId,
                                int seatNumber)
            throws SQLException {

        String sql =
                "select booking_id " +
                "from bookings " +
                "where bus_id = ? " +
                "and seat_number = ? " +
                "and status = 'CONFIRMED'";

        PreparedStatement ps =
                con.prepareStatement(sql);

        ps.setInt(1, busId);
        ps.setInt(2, seatNumber);

        ResultSet rs =
                ps.executeQuery();

        return rs.next();
    }


    public int createBooking(Connection con,
                             int busId,
                             int passengerId,
                             int seatNumber)
            throws SQLException {

        String sql =
                "insert into bookings " +
                "(bus_id, passenger_id, " +
                "seat_number, status) " +
                "values (?, ?, ?, 'CONFIRMED')";

        PreparedStatement ps =
                con.prepareStatement(
                        sql,
                        Statement.RETURN_GENERATED_KEYS);

        ps.setInt(1, busId);
        ps.setInt(2, passengerId);
        ps.setInt(3, seatNumber);

        ps.executeUpdate();

        ResultSet rs =
                ps.getGeneratedKeys();

        if (rs.next()) {
            return rs.getInt(1);
        }

        return -1;
    }


    public ResultSet getBooking(Connection con,
                                int bookingId)
            throws SQLException {

        String sql =
                "select * from bookings " +
                "where booking_id = ?";

        PreparedStatement ps =
                con.prepareStatement(sql);

        ps.setInt(1, bookingId);

        return ps.executeQuery();
    }


    public void cancelBooking(Connection con,
                              int bookingId)
            throws SQLException {

        String sql =
                "update bookings " +
                "set status = 'CANCELLED' " +
                "where booking_id = ?";

        PreparedStatement ps =
                con.prepareStatement(sql);

        ps.setInt(1, bookingId);

        ps.executeUpdate();
    }


    public void viewTicket(int bookingId) {

import java.sql.Connection;
import java.sql.ResultSet;
import java.sql.SQLException;

public class BookingService {

    private BusDAO busDAO =
            new BusDAO();

    private PassengerDAO passengerDAO =
            new PassengerDAO();

    private BookingDAO bookingDAO =
            new BookingDAO();


    public void bookTicket(int busId,
                           int seatNumber,
                           String name,
                           int age,
                           String gender,
                           String phone) {

        Connection con = null;

        try {

            con = DBConnection.getConnection();

            con.setAutoCommit(false);


            int availableSeats =
                    busDAO.getAvailableSeats(
                            con,
                            busId);


            if (availableSeats == -1) {

                System.out.println(
                        "Bus not found.");

                con.rollback();

                return;
            }


            int totalSeats =
                    busDAO.getTotalSeats(
                            con,
                            busId);


            System.out.println(
                    "\nTotal Seats     : " +
                    totalSeats);

            System.out.println(
                    "Available Seats : " +
                    availableSeats);


            if (availableSeats <= 0) {

                System.out.println(
                        "No seats available.");

                con.rollback();

                return;
            }


            if (seatNumber < 1 ||
                seatNumber > totalSeats) {

                System.out.println(
                        "Invalid seat number.");

                System.out.println(
                        "Please enter seat number " +
                        "between 1 and " +
                        totalSeats);

                con.rollback();

                return;
            }


            boolean booked =
                    bookingDAO.isSeatBooked(
                            con,
                            busId,
                            seatNumber);


            if (booked) {

                System.out.println(
                        "Seat " +
                        seatNumber +
                        " is already booked.");

                con.rollback();

                return;
            }


            int passengerId =
                    passengerDAO.addPassenger(
                            con,
                            name,
                            age,
                            gender,
                            phone);


            int bookingId =
                    bookingDAO.createBooking(
                            con,
                            busId,
                            passengerId,
                            seatNumber);


            busDAO.decreaseSeat(
                    con,
                    busId);


            con.commit();


            System.out.println(
                    "\n================================");

            System.out.println(
                    "       BOOKING SUCCESSFUL");

            System.out.println(
                    "================================");

            System.out.println(
                    "Booking ID : " +
                    bookingId);

            System.out.println(
                    "Passenger ID : " +
                    passengerId);

            System.out.println(
                    "Bus ID : " +
                    busId);

            System.out.println(
                    "Seat Number : " +
                    seatNumber);

            System.out.println(
                    "Status : CONFIRMED");

            System.out.println(
                    "================================");


        } catch (SQLException e) {

            try {

                if (con != null) {
                    con.rollback();
                }

            } catch (SQLException ex) {
                ex.printStackTrace();
            }

            System.out.println(
                    "Booking failed.");

            e.printStackTrace();


        } finally {

            try {

                if (con != null) {
                    con.close();
                }

            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }


    public void cancelTicket(int bookingId) {

        Connection con = null;

        try {

            con = DBConnection.getConnection();

            con.setAutoCommit(false);


            ResultSet rs =
                    bookingDAO.getBooking(
                            con,
                            bookingId);


            if (!rs.next()) {

                System.out.println(
                        "Booking not found.");

                con.rollback();

                return;
            }


            int busId =
                    rs.getInt("bus_id");


            int seatNumber =
                    rs.getInt("seat_number");


            String status =
                    rs.getString("status");


            if (status.equals("CANCELLED")) {

                System.out.println(
                        "Ticket already cancelled.");

                con.rollback();

                return;
            }


            bookingDAO.cancelBooking(
                    con,
                    bookingId);


            busDAO.increaseSeat(
                    con,
                    busId);


            con.commit();


            System.out.println(
                    "\nTicket cancelled successfully.");

            System.out.println(
                    "Seat " +
                    seatNumber +
                    " is now available.");


        } catch (SQLException e) {

            try {

                if (con != null) {
                    con.rollback();
                }

            } catch (SQLException ex) {
                ex.printStackTrace();
            }

            System.out.println(
                    "Cancellation failed.");

            e.printStackTrace();


        } finally {

            try {

                if (con != null) {
                    con.close();
                }

            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }
}
